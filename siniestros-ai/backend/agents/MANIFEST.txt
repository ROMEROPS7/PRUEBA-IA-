DATABASE SCHEMA
===============

Required table for logging:

CREATE TABLE agent_logs (
    id VARCHAR(36) PRIMARY KEY,
    agent_id VARCHAR(50),
    agent_name VARCHAR(100),
    siniestro_id VARCHAR(50),
    action VARCHAR(100),
    input_data TEXT,
    output_data TEXT,
    duration_ms FLOAT,
    status VARCHAR(20),
    error_message TEXT,
    timestamp DATETIME,
    INDEX idx_siniestro (siniestro_id),
    INDEX idx_timestamp (timestamp),
    INDEX idx_agent (agent_id)
);
